Mostrar Capacidade Operacional const divEst = document.getElementById('printEstacoes'); divEst.innerHTML = ""; dadosApp.estacoes.forEach(est => { divEst.innerHTML += `
${est.nome}: ${est.qtd} un.
`; }); divEst.innerHTML += `
Total Geral Disponível: ${totalDisponivel} Notebooks
`; // 2 e 3. Construir Grade Geral de Uso e Mapear Saturações const corpoGeral = document.getElementById('printCorpoGradeGeral'); corpoGeral.innerHTML = ""; const alertasDiv = document.getElementById('printAlertas'); alertasDiv.innerHTML = ""; let temConflitoGlobal = false; for (let i = 0; i < dadosApp.modeloAulas; i++) { const tr = document.createElement('tr'); const tdNum = document.createElement('td'); tdNum.className = "border border-gray-300 p-2 font-bold bg-gray-50"; tdNum.innerText = `${i+1}ª Aula`; tr.appendChild(tdNum); diasSemana.forEach(dia => { let notesNestaAula = 0; let turmasUtilizando = []; Object.keys(dadosApp.turmas).forEach(turma => { if (dadosApp.turmas[turma][dia] && dadosApp.turmas[turma][dia][i]) { notesNestaAula += 1; turmasUtilizando.push(turma); } }); const td = document.createElement('td'); td.className = "border border-gray-300 p-2"; const numCarrinhosNecessarios = turmasUtilizando.length; const numCarrinhosDisponiveis = dadosApp.estacoes.length; if (numCarrinhosNecessarios > numCarrinhosDisponiveis) { td.className += " bg-red-100 text-red-700 font-bold"; td.innerHTML = `⚠️ ${numCarrinhosNecessarios} salas
(${turmasUtilizando.join(', ')})`; alertasDiv.innerHTML += `
❌ ${dia.toUpperCase()} (${i+1}ª Aula): Saturação de infraestrutura! Existem ${numCarrinhosNecessarios} turmas tentando usar notebooks simultaneamente, mas a escola possui apenas ${numCarrinhosDisponiveis} carrinho(s). Faltam equipamentos para atender as turmas: ${turmasUtilizando.join(', ')}.
`; temConflitoGlobal = true; } else if (numCarrinhosNecessarios > 0) { td.className += " bg-emerald-50 text-emerald-800 font-medium"; td.innerHTML = `💻 ${numCarrinhosNecessarios} sala(s)
${turmasUtilizando.join(', ')}`; } else { td.className += " text-gray-300"; td.innerText = "-"; } tr.appendChild(td); }); corpoGeral.appendChild(tr); } if (!temConflitoGlobal) { alertasDiv.innerHTML = `
✅ Todos os horários estão compatíveis com a capacidade de carrinhos/notebooks instalada. Sem pontos de saturação identificados!
`; } // Força a tela a rolar para o topo do relatório para o usuário ver o resultado window.scrollTo({ top: 0, behavior: 'smooth' }); } function fecharRelatorio() { document.getElementById('secaoHorarios').closest('main').classList.remove('hidden'); document.querySelector('header').classList.remove('hidden'); document.getElementById('areaRelatorio').classList.add('hidden'); }
